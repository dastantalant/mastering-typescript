async function fetchDictionary() {

    const findAllExtended = await SaimaTemplateCategoryService.findAllExtended();

    dictionaries.value = {
      channels: enums.findAllRegistrarChanel,
      regions: enums.getAllRegion,

      documentTypes: enums.getAllDocumentType,
      saimaSubEditData: enums.getAllSaimaSubEditData,
      tariffs: getAllConnectionDataTypes.tariffs,
      routers: getAllConnectionDataTypes.routers,
      consoles: getAllConnectionDataTypes.consoles,
      saimaTemplateCategories: findAllExtended,
    };

    buildSaimaStateMap(enums.getAllSaimaState);
    buildPhotoTypesMap(enums.getAllSaimaPhotoType);
    buildTemplateMap(findAllExtended);

  } catch (error: any) {
    toast.add({
      severity: 'error',
      summary: 'Ошибка загрузки справочников',
      detail: error.message,
      life: 5000
    });

function buildSaimaStateMap(states: Enum[]) {
  for (const state of states) {
    maps.value.saimaStateMap.set(state.value, state);
  }
}

function buildPhotoTypesMap(photoTypes: Enum[]) {
  for (const element of photoTypes) {
    const photo: Photo = {
      id: element.code,
      name: element.name,
      type: element.value,
    };
    photoTypesList.value.push(photo);
    maps.value.photoTypesMap.set(element.value, photo);
  }
}

function buildTemplateMap(categories: SaimaTemplateCategory[]) {
  for (const category of categories) {
    maps.value.saimaTemplatesByCategory[category.id] = category.saimaTemplates || [];
    for (const template of category.saimaTemplates) {
      maps.value.saimaTemplateMap.set(template.id, template);
    }
  }
}

async function fetchData(params: FetchDataParams) {
  const filter: SaimaAgentFilter = {
    page: params.page,
    size: params.rows,
    ...SaimaAgentService.createSaimaAgentFilter(params.filters)
  };

  try {
    const data: Page<ShortSaimaAgentDTO> = await SaimaAgentService.findAll(filter);
    return {items: data.content || [], totalRecords: data.totalElements || 0};
  } catch (error: any) {
    toast.add({severity: 'error', summary: 'Ошибка', detail: error.message, life: 5000});
    return {items: [], totalRecords: 0};
  }
}

function buildPhotoItems(uploads: SaimaAgentUploads[] = []): Photo[] {
  return uploads
      .filter(upload => upload.saimaPhotoType)
      .map(upload => {
        const photo = maps.value.photoTypesMap.get(upload.saimaPhotoType);
        return photo ? {...photo, url: upload.signedUrl} : null;
      })
      .filter(Boolean);
}

function isAgentEditable(agent: SaimaAgent): boolean {
  return agent.editable && !!agent.msisdn && MSISDN_REGEXP.test(agent.msisdn);
}

async function dialogHandleSave(context: DialogContext<SaimaAgent>) {
  dialogLoading.value = true;
  dialogMessage.value = {};

  try {
    await SaimaAgentService.put(
        context.editData,
        SaimaAgentMapper.toUpdateDTO(context.editData)
    );

    dialogVisible.value = false;
    toast.add({
      severity: 'success',
      summary: 'Успешно',
      detail: 'Данные сохранены',
      life: 3000
    });

    tableDataTableRef.value?.refresh();
  } catch (error: any) {
    dialogMessage.value = {
      text: getErrorMessage(error),
      severity: 'error',
    } as Record<string, string>;
    console.error("Failed to save data:", error);
  } finally {
    dialogLoading.value = false;
  }
}

function getErrorMessage(error: any): string {
  const status = error?.response?.status;
  const responseData = error?.response?.data;

  const errorMessages = {
    400: responseData?.message || responseData?.error || 'Неверные данные. Проверьте заполненные поля.',
    401: 'Ошибка авторизации. Пожалуйста, войдите в систему заново.',
    403: 'У вас нет прав для выполнения этого действия.',
    404: 'Сайма Агент не найден.',
    409: responseData?.message || 'Конфликт данных. Возможно, такая запись уже существует.',
    500: 'Произошла внутренняя ошибка сервера. Попробуйте позже.',
  } as Record<number, string>;

  if (status && errorMessages[status]) {
    return errorMessages[status];
  }

  if (error instanceof TypeError && error.message === 'Failed to fetch') {
    return 'Ошибка сети. Проверьте ваше интернет-соединение.';
  }

  return error?.message || 'Произошла непредвиденная ошибка.';
}

async function fetchAndConfirmPassportData(context: DialogContext<SaimaAgent>) {
  const {inn, documentId} = context.editData;

  if (!inn || !documentId) {
    toast.add({
      severity: 'warn',
      summary: 'Внимание',
      detail: 'Введите ИНН и номер документа',
      life: 3000
    });
    return;
  }

  isFetchingPassportData.value = true;

  try {
    passportDataToConfirm.value = await PassportService.getPassportData(inn, documentId);
    isPassportConfirmDialogVisible.value = true;
  } catch (error: any) {
    const detail = error?.response?.status === 404
        ? 'Данные по ПИН не найдены.'
        : 'Не удалось получить данные паспорта.';

    toast.add({severity: 'error', summary: 'Ошибка', detail, life: 5000});
  } finally {
    isFetchingPassportData.value = false;
  }
}

function applyPassportData() {
  if (!passportDataToConfirm.value || !dataEditorDialogRef.value) {
    return;
  }

  const response = passportDataToConfirm.value;
  const birthDate = parseDate(response.dateOfBirth);
  const issuedDate = parseDate(response.issuedDate);

  const updates = {
    surname: response.surname,
    firstName: response.name,
    middleName: response.patronymic,
    birthDate,
    issuedDate,
    authority: response.passportAuthority,
    address: response.address
  };

  dataEditorDialogRef.value.updateEditData(updates);

  setTimeout(() => {
    dataEditorDialogRef.value?.revalidateAllFields();
  }, 0);

  toast.add({
    severity: 'success',
    summary: 'Успешно',
    detail: 'Данные применены',
    life: 3000
  });

  isPassportConfirmDialogVisible.value = false;
}

function parseDate(dateString: string): Date | null {
  if (!dateString) return null;
  return DateTime.fromFormat(dateString, 'dd.MM.yyyy').toJSDate();
}

function generatePassportMessage(data: PassportResponse): string {
  return `
    <p><strong>ФИО:</strong> ${data.surname} ${data.name} ${data.patronymic}</p>
    <p><strong>Дата рождения:</strong> ${data.dateOfBirth}</p>
    <p><strong>Дата выдачи:</strong> ${data.issuedDate}</p>
    <p><strong>Орган:</strong> ${data.passportAuthority}</p>
    <p><strong>Адрес регистрации:</strong> ${data.address}</p>
  `;
}

    });

    const getContext = (): DialogContext<T> => ({
        editData: editData.value,
        formData: formData.value,
        isEditing: isEditing,
    });

    const applyDefaults = (data: T): T => {
        const result: Record<string, any> = {...data};
        for (const field of formFieldsOnly.value) {
            if (field.defaultValue !== undefined && (result[field.field] === undefined || result[field.field] === null)) {
                result[field.field] = typeof field.defaultValue === 'function' ? field.defaultValue() : JSON.parse(JSON.stringify(field.defaultValue));
            }
        }
        return result as T;
    };

    watch(() => props.initialData, (newData) => {
        const baseData = newData ? JSON.parse(JSON.stringify(newData)) : {} as T;
        formData.value = applyDefaults(baseData);
    }, {immediate: true, deep: true});

    watch(() => props.visible, (isVisible) => {
        if (isVisible) {
            if (props.config.photos?.length) {
                activePhotoId.value = props.config.photos[0].id;
            }
        } else {
            isEditing.value = false;
            activePhotoId.value = '';
        }
    });

    const revalidateConditionalFields = () => {
        for (const field of formFieldsOnly.value) {
            if (field.requiredCondition || field.validation?.validator || field.warning) {
                validateField(field);
                checkFieldWarning(field);
            }
        }
    };

    const validateField = (fieldDef: FormField<T>): boolean => {
        const {field, label, validation} = fieldDef;
        const value = editData.value[field];
        let error = '';
        const isRequired = fieldDef.required && (!fieldDef.requiredCondition || fieldDef.requiredCondition(getContext()));

        if (isRequired && (value === null || value === undefined || value === '' || (Array.isArray(value) && value.length === 0))) {
            error = typeof fieldDef.required === 'string' ? fieldDef.required : DefaultMessages.RequiredField.replace('{label}', label);
        } else if (value !== null && value !== undefined && value !== '') {
            if (validation) {
                if (validation.validator) {
                    const customError = validation.validator(value, getContext());
                    if (customError) error = customError;
                }
                if (!error) {
                    if (validation.pattern && !validation.pattern.test(String(value))) {
                        error = validation.patternMessage || DefaultMessages.InvalidFormat;
                    } else if (validation.len !== undefined && String(value).length !== validation.len) {
                        error = validation.lenMessage || DefaultMessages.LengthMustBe.replace('{label}', label).replace('{len}', String(validation.len));
                    } else if (typeof value === 'number') {
                        if (validation.min !== undefined && value < validation.min) {
                            error = validation.minMessage || DefaultMessages.MinLength.replace('{min}', String(validation.min));
                        } else if (validation.max !== undefined && value > validation.max) {
                            error = validation.maxMessage || DefaultMessages.MaxLength.replace('{max}', String(validation.max));
                        }
                    }
                }
            }
        }

        if (error) {
            errors.value[field] = error;
            return false;
        } else {
            delete errors.value[field];
            return true;
        }
    };

    const checkFieldWarning = (fieldDef: FormField<T>): void => {
        const {field, warning} = fieldDef;
        if (!warning) {
            delete warnings.value[field];
            return;
        }
        const value = editData.value[field];
        const warningMessage = warning(value, getContext());
        if (warningMessage) {
            warnings.value[field] = warningMessage;
        } else {
            delete warnings.value[field];
        }
    };

    const handleFieldBlur = (fieldDef: FormField<T>) => {
        validateField(fieldDef);
        checkFieldWarning(fieldDef);
        revalidateConditionalFields();
    };

    const onFieldValueUpdate = (fieldDef: FormField<T>, newValue: any) => {
        validateField(fieldDef);
        checkFieldWarning(fieldDef);
        for (const field of formFieldsOnly.value) {
            if (field.type === FieldType.DependentSelect && field.masterField === fieldDef.field) {
                editData.value[field.field] = null;
                validateField(field);
            }
        }
        revalidateConditionalFields();
        if (fieldDef.onChange) {
            fieldDef.onChange(newValue, getContext());
        }
    };

    const validateForm = () => {
        let isValid = true;
        for (const field of formFieldsOnly.value) {
            if (isEditing.value && field.editable !== false) {
                const isDisplayed = !field.displayCondition || field.displayCondition(getContext());
                if (isDisplayed && !validateField(field)) {
                    isValid = false;
                }
            }
        }
        return isValid;
    };

    const startEditing = async () => {
        if (isStartingEdit.value) return;
        isStartingEdit.value = true;
        try {
            let dataForEditing = formData.value;
            if (props.config.onBeforeEdit) {
                const updatedData = await props.config.onBeforeEdit(getContext());
                if (updatedData) {
                    formData.value = JSON.parse(JSON.stringify(updatedData));
                    dataForEditing = updatedData;
                }
            }
            editData.value = JSON.parse(JSON.stringify(dataForEditing));
            isEditing.value = true;
            for (const field of formFieldsOnly.value) {
                checkFieldWarning(field)
            }
            validateForm();
        } catch (error) {
            console.error('Ошибка при подготовке к редактированию:', error);
        } finally {
            isStartingEdit.value = false;
        }
    };

    const cancelEditing = () => {
        isEditing.value = false;
        errors.value = {};
        warnings.value = {};
        editData.value = {} as T;
    };

    const handleSave = () => {
        if (validateForm()) {
            emit('save', getContext());
        }
    };

    const handleClose = () => {
        emit('cancel');
        onHide();
    };

    const onHide = () => {
        emit('update:visible', false);
    };

    const isSaveDisabled = computed(() => {
        if (!isEditing.value) return false;
        if (Object.keys(errors.value).length > 0) return true;
        return formFieldsOnly.value.some(field => {
            const isRequired = field.required && (!field.requiredCondition || field.requiredCondition(getContext()));
            if (isRequired) {
                const value = editData.value[field.field];
                return value === null || value === undefined || value === '' || (Array.isArray(value) && value.length === 0);
            }
            return false;
        });
    });

    const displayData = computed(() => {
        if (!formData.value) return {};
        const result: Record<string, any> = {...formData.value};
        for (const field of formFieldsOnly.value) {
            const value = formData.value[field.field];
            if (value === undefined || value === null || value === '' || value === 'null') {
                result[field.field] = DefaultMessages.Empty;
                continue;
            }
            if (field.type === FieldType.Select || field.type === FieldType.Dropdown) {
                const options = field.options || [];
                const optionValueKey = field.optionValue || 'value';
                const optionLabelKey = field.optionLabel || 'label';
                const foundOption = options.find((opt: any) => (typeof opt === 'object' && opt) ? opt[optionValueKey] === value : opt === value);
                if (foundOption) {
                    result[field.field] = (typeof foundOption === 'object' && foundOption) ? foundOption[optionLabelKey] : foundOption;
                }
            } else if (field.type === FieldType.MultiSelect) {
                const options = field.options || [];
                const optionValueKey = field.optionValue || 'value';
                const optionLabelKey = field.optionLabel || 'label';
                if (Array.isArray(value) && value.length > 0) {
                    result[field.field] = value.map(val => {
                        const foundOption = options.find((opt: any) => (typeof opt === 'object' && opt) ? opt[optionValueKey] === val : opt === val);
                        return foundOption ? ((typeof foundOption === 'object' && foundOption) ? foundOption[optionLabelKey] : foundOption) : val;
                    }).join(', ');
                } else {
                    result[field.field] = DefaultMessages.Empty;
                }
            } else if (field.type === FieldType.DependentSelect) {
                const masterValue = formData.value[field.masterField];
                if (masterValue) {
                    const options = field.options[masterValue] || [];
                    const optionValueKey = field.optionValue || 'value';
                    const optionLabelKey = field.optionLabel || 'label';
                    const foundOption = options.find((opt: any) => (typeof opt === 'object' && opt !== null) ? opt[optionValueKey] === value : opt === value);
                    if (foundOption) {
                        result[field.field] = (typeof foundOption === 'object' && foundOption) ? foundOption[optionLabelKey] : foundOption;
                    }
                }
            } else if (field.type === FieldType.Date && value) {
                try {
                    const format = field.dateFormat ? field.dateFormat : (field.showTime ? 'dd.MM.yyyy HH:mm' : 'dd.MM.yyyy');
                    result[field.field] = DateTime.fromISO(value).toFormat(format);
                } catch (_) {
                    result[field.field] = value;
                }
            }
        }
        return result;
    });

    const dependentOptions = computed(() => {
        if (!formData.value || Object.keys(formData.value).length === 0) return {};
        const result: Record<string, any> = {};
        for (const field of formFieldsOnly.value) {
            if (field.type === FieldType.DependentSelect) {
                const masterValue = editData.value[field.masterField];
                result[field.field] = masterValue ? field.options[masterValue] || [] : [];
            }
        }
        return result;
    });

    const getFieldComponent = (field: FormField<T>): Component => {
        switch (field.type) {
            case FieldType.Text:
                return InputText;
            case FieldType.Number:
                return InputNumber;
            case FieldType.Select:
            case FieldType.Dropdown:
            case FieldType.DependentSelect:
                return Select;
            case FieldType.MultiSelect:
                return MultiSelect;
            case FieldType.Date:
                return DatePicker;
            case FieldType.Tag:
                return Tag;
            case FieldType.Textarea:
                return Textarea;
            default:
                return InputText;
        }
    };

    const getFieldProps = (field: FormField<T>) => {
        const isFullWidth = ![FieldType.Date, FieldType.Tag].includes(field.type);
        const commonProps = {
            id: field.field,
            disabled: isFieldDisabled(field),
            class: {
                'p-invalid': errors.value[field.field],
                'p-warn': warnings.value[field.field] && !errors.value[field.field],
                'full-width-field': isFullWidth,
            }
        };
        switch (field.type) {
            case FieldType.Select:
            case FieldType.Dropdown:
            case FieldType.MultiSelect:
                return {
                    ...commonProps,
                    options: field.options,
                    optionLabel: field.optionLabel,
                    optionValue: field.optionValue,
                    filter: field.filter || false,
                    showClear: field.showClear ?? true
                };
            case FieldType.DependentSelect:
                return {
                    ...commonProps,
                    options: dependentOptions.value[field.field],
                    optionLabel: field.optionLabel,
                    optionValue: field.optionValue,
                    showClear: field.showClear ?? true
                };
            case FieldType.Textarea:
                return {...commonProps, autoResize: field.autoResize ?? true, rows: field.rows ?? 5};
            case FieldType.Date:
                return {...commonProps, dateFormat: replaceDateFormat(field.dateFormat), showTime: field.showTime};
            default:
                return commonProps;
        }
    };

    const isFieldDisabled = (field: FormField<T>): boolean => {
        if (!isEditing.value) return true;
        if (field.editable === false) return true;
        if (field.disabled) return true;
        if (field.type === FieldType.DependentSelect && !editData.value[field.masterField]) return true;
        return !!(field.disabledCondition?.(getContext()));
    };

    const isFieldRequired = (fieldDef: FormField<T>): boolean => {
        if (!isEditing.value || !fieldDef.required) return false;
        return !fieldDef.requiredCondition || fieldDef.requiredCondition(getContext());
    };

    const isActionVisible = (btn: DialogAction<T>): boolean => {
        if (!btn.visible) return true;
        if (typeof btn.visible === 'function') return btn.visible(getContext());
        if (btn.visible === ActionVisibility.Always) return true;
        if (btn.visible === ActionVisibility.View) return !isEditing.value;
        if (btn.visible === ActionVisibility.Edit) return isEditing.value;
        return false;
    };

    const getViewModeClasses = (field: FormField<T>) => ({
        'view-mode-field': true,
        'auto-width-field': [FieldType.Date, FieldType.Tag].includes(field.type),
        'view-mode-textarea': field.type === FieldType.Textarea,
    });

    const getViewModeStyle = (field: FormField<T>) => {
        if (field.type === FieldType.Textarea) {
            const rows = (field as any).rows || 3;
            return {minHeight: `${rows * 1.6}rem`};
        }
        return {};
    };

    const revalidateAllFields = () => {
        for (const field of formFieldsOnly.value) {
            validateField(field);
            checkFieldWarning(field);
        }
    };

    const updateEditData = (updates: Partial<T>) => {
        if (isEditing.value) {
            editData.value = {...editData.value, ...updates};
        }
    };

    return {
        isEditing,
        formData,
        editData,
        errors,
        warnings,
        isStartingEdit,
        activePhotoId,
        isRecordEditable,
        isSaveDisabled,
        displayData,
        getContext,
        startEditing,
        cancelEditing,
        handleSave,
        handleClose,
        onHide,
        handleFieldBlur,
        onFieldValueUpdate,
        isFieldRequired,
        isFieldDisabled,
        isActionVisible,
        getFieldComponent,
        getFieldProps,
        getViewModeClasses,
        getViewModeStyle,
        revalidateAllFields,
        updateEditData,
    };
}
